import type { ChargingService } from '../../charging/application/ChargingService.ts';
import type { AuthorizationService } from '../../charging/application/ports.ts';
import { connectorIdOf } from '../../charging/domain/Connector.ts';

// OCPP 1.6J call frame: [2, uniqueId, action, payload]; result: [3, uniqueId, payload].
export type OcppCall = [2, string, string, Record<string, any>];
export type OcppResult = [3, string, Record<string, unknown>];

// Anti-corruption layer. OCPP vocabulary (StatusNotification, transactionId, meterStart, sampledValue)
// stops here; only ubiquitous-language commands go inward. One ACL instance per charger connection.
export class OcppAcl {
  private readonly chargerId: string;
  private readonly charging: ChargingService;
  private readonly authorization: AuthorizationService;
  private nextTransactionId: number;

  constructor(
    chargerId: string,
    charging: ChargingService,
    authorization: AuthorizationService,
    options: { firstTransactionId?: number } = {},
  ) {
    this.chargerId = chargerId;
    this.charging = charging;
    this.authorization = authorization;
    this.nextTransactionId = options.firstTransactionId ?? 991; // -> S-991 for the canonical afternoon
  }

  async handle(frame: OcppCall): Promise<OcppResult> {
    const [, uniqueId, action, payload] = frame;
    const result = await this.dispatch(action, payload);
    return [3, uniqueId, result];
  }

  private async dispatch(action: string, p: Record<string, any>): Promise<Record<string, unknown>> {
    switch (action) {
      case 'BootNotification':
        return { status: 'Accepted', currentTime: new Date().toISOString(), interval: 300 };

      case 'Heartbeat':
        return { currentTime: new Date().toISOString() };

      case 'Authorize':
        return { idTagInfo: { status: (await this.authorization.isAuthorized(p.idTag)) ? 'Accepted' : 'Invalid' } };

      case 'StatusNotification': {
        if (p.status !== 'Faulted') return {}; // Available/Preparing/... carry no command in MVP1
        await this.charging.reportFault({
          chargerId: this.chargerId,
          connectorId: p.connectorId > 0 ? connectorIdOf(this.chargerId, p.connectorId) : undefined,
          faultCode: p.errorCode,
          at: p.timestamp,
        });
        return {};
      }

      case 'StartTransaction': {
        const transactionId = this.nextTransactionId++;
        const events = await this.charging.startCharging({
          sessionId: sessionIdFor(transactionId),
          connectorId: connectorIdOf(this.chargerId, p.connectorId),
          idTag: p.idTag,
          meterStartWh: p.meterStart,
          at: p.timestamp,
        });
        const rejected = events.find((e) => e.type === 'ChargingStartRejected');
        return {
          transactionId,
          idTagInfo: { status: rejected ? (rejected.reason === 'Unauthorized' ? 'Invalid' : 'Blocked') : 'Accepted' },
        };
      }

      case 'MeterValues': {
        const meterWh = energyRegisterWh(p.meterValue);
        if (meterWh === undefined || p.transactionId === undefined) return {};
        await this.charging.reportMeterValue({
          sessionId: sessionIdFor(p.transactionId),
          meterWh,
          at: p.meterValue[0].timestamp,
        });
        return {};
      }

      case 'StopTransaction': {
        const sessionId = sessionIdFor(p.transactionId);
        // The final register value is just another meter reading (R3), then the stop (R4).
        await this.charging.reportMeterValue({ sessionId, meterWh: p.meterStop, at: p.timestamp });
        await this.charging.stopCharging({ sessionId, stopReason: p.reason ?? 'Local', at: p.timestamp });
        return { idTagInfo: { status: 'Accepted' } };
      }

      default:
        throw new Error(`OCPP action not supported: ${action}`);
    }
  }
}

function sessionIdFor(transactionId: number): string {
  return `S-${transactionId}`;
}

// MeterValues carries a list of samples; we only care about the cumulative energy register in Wh.
function energyRegisterWh(meterValue: Array<{ sampledValue: Array<Record<string, string>> }> | undefined): number | undefined {
  for (const entry of meterValue ?? []) {
    for (const sample of entry.sampledValue ?? []) {
      const measurand = sample.measurand ?? 'Energy.Active.Import.Register';
      if (measurand !== 'Energy.Active.Import.Register') continue;
      const value = Number(sample.value);
      return sample.unit === 'kWh' ? Math.round(value * 1000) : Math.round(value);
    }
  }
  return undefined;
}

"""Anti-corruption layer. OCPP vocabulary (StatusNotification, transactionId, meterStart, sampledValue)
stops here; only ubiquitous-language commands go inward. One ACL instance per charger connection.

OCPP 1.6J call frame: [2, uniqueId, action, payload]; result: [3, uniqueId, payload].
"""
from datetime import datetime, timezone
from typing import Any

from charging.application.charging_service import ChargingService
from charging.application.commands import ReportFault, ReportMeterValue, StartCharging, StopCharging
from charging.application.ports import AuthorizationService
from charging.domain.connector import connector_id_of
from charging.domain.events import ChargingStartRejected

OcppCall = list  # [2, unique_id, action, payload]
OcppResult = list  # [3, unique_id, payload]


class OcppAcl:
    def __init__(self, charger_id: str, charging: ChargingService, authorization: AuthorizationService, first_transaction_id: int = 991) -> None:
        self._charger_id = charger_id
        self._charging = charging
        self._authorization = authorization
        self._next_transaction_id = first_transaction_id  # -> S-991 for the canonical afternoon

    def handle(self, frame: OcppCall) -> OcppResult:
        _, unique_id, action, payload = frame
        return [3, unique_id, self._dispatch(action, payload)]

    def _dispatch(self, action: str, p: dict[str, Any]) -> dict[str, Any]:
        now = datetime.now(timezone.utc).isoformat()
        if action == "BootNotification":
            return {"status": "Accepted", "currentTime": now, "interval": 300}
        if action == "Heartbeat":
            return {"currentTime": now}
        if action == "Authorize":
            return {"idTagInfo": {"status": "Accepted" if self._authorization.is_authorized(p["idTag"]) else "Invalid"}}
        if action == "StatusNotification":
            if p.get("status") != "Faulted":
                return {}  # Available/Preparing/... carry no command in MVP1
            connector_id = connector_id_of(self._charger_id, p["connectorId"]) if p.get("connectorId", 0) > 0 else None
            self._charging.report_fault(ReportFault(charger_id=self._charger_id, connector_id=connector_id, fault_code=p["errorCode"], at=p["timestamp"]))
            return {}
        if action == "StartTransaction":
            transaction_id = self._next_transaction_id
            self._next_transaction_id += 1
            events = self._charging.start_charging(
                StartCharging(
                    session_id=_session_id_for(transaction_id),
                    connector_id=connector_id_of(self._charger_id, p["connectorId"]),
                    id_tag=p["idTag"],
                    meter_start_wh=p["meterStart"],
                    at=p["timestamp"],
                )
            )
            rejected = next((e for e in events if isinstance(e, ChargingStartRejected)), None)
            status = "Accepted" if rejected is None else ("Invalid" if rejected.reason == "Unauthorized" else "Blocked")
            return {"transactionId": transaction_id, "idTagInfo": {"status": status}}
        if action == "MeterValues":
            meter_wh = _energy_register_wh(p.get("meterValue"))
            if meter_wh is None or "transactionId" not in p:
                return {}
            self._charging.report_meter_value(ReportMeterValue(session_id=_session_id_for(p["transactionId"]), meter_wh=meter_wh, at=p["meterValue"][0]["timestamp"]))
            return {}
        if action == "StopTransaction":
            session_id = _session_id_for(p["transactionId"])
            # The final register value is just another meter reading (R3), then the stop (R4).
            self._charging.report_meter_value(ReportMeterValue(session_id=session_id, meter_wh=p["meterStop"], at=p["timestamp"]))
            self._charging.stop_charging(StopCharging(session_id=session_id, stop_reason=p.get("reason", "Local"), at=p["timestamp"]))
            return {"idTagInfo": {"status": "Accepted"}}
        raise ValueError(f"OCPP action not supported: {action}")


def _session_id_for(transaction_id: int) -> str:
    return f"S-{transaction_id}"


def _energy_register_wh(meter_value: list[dict[str, Any]] | None) -> int | None:
    """MeterValues carries a list of samples; we only care about the cumulative energy register in Wh."""
    for entry in meter_value or []:
        for sample in entry.get("sampledValue", []):
            if sample.get("measurand", "Energy.Active.Import.Register") != "Energy.Active.Import.Register":
                continue
            value = float(sample["value"])
            return round(value * 1000) if sample.get("unit") == "kWh" else round(value)
    return None

import Fastify, { type FastifyInstance } from 'fastify';
import type { App } from '../../app.ts';
import { OcppAcl, type OcppCall } from '../ocpp/OcppAcl.ts';
import { ParkingSession } from '../../parking/ParkingSession.ts';
import { newId } from '../../shared/Ids.ts';
import { InvalidStateError } from '../../charging/domain/errors.ts';

// Inbound HTTP adapter. Every handler does three things and nothing else:
// parse the request -> run ONE application-service call inside the unit of work -> map the result to a status code.
// No `if (session.status === 'Charging')` here: that is R1-R4 and it lives in the aggregate.
export function buildServer(app: App, options: { logger?: boolean | object } = {}): FastifyInstance {
  const server = Fastify({
    logger: options.logger ?? false,
    genReqId: (req) => (req.headers['x-correlation-id'] as string | undefined) ?? newId(),
  });
  const acls = new Map<string, OcppAcl>([[app.chargerId, app.acl]]);
  const parking = new Map<string, ParkingSession>();
  const cid = (req: { id: string }) => ({ correlationId: req.id, causationId: req.id });

  server.get('/health', async () => ({ ok: true, pendingOutbox: (await app.chargingOutbox.pending()).length + (await app.opsOutbox.pending()).length }));

  // ---- Charging commands -----------------------------------------------------------------
  server.post<{ Body: { sessionId?: string; connectorId: string; idTag: string; meterStartWh?: number; at: string } }>('/sessions/start', async (req, reply) => {
    const b = req.body;
    const sessionId = b.sessionId ?? `S-${newId().slice(0, 8)}`;
    const events = await app.uow.run(() =>
      app.charging.startCharging({ sessionId, connectorId: b.connectorId, idTag: b.idTag, meterStartWh: b.meterStartWh ?? 0, at: b.at, ...cid(req) }),
    );
    const rejected = events.find((e) => e.type === 'ChargingStartRejected');
    if (rejected) return reply.code(rejected.reason === 'Unauthorized' ? 403 : 409).send({ error: 'ChargingStartRejected', reason: rejected.reason });
    return reply.code(201).send({ sessionId, connectorId: b.connectorId, status: 'Charging' });
  });

  server.post<{ Params: { id: string }; Body: { meterWh: number; at: string } }>('/sessions/:id/meter', async (req, reply) => {
    const events = await app.uow.run(() => app.charging.reportMeterValue({ sessionId: req.params.id, meterWh: req.body.meterWh, at: req.body.at, ...cid(req) }));
    const rejected = events.find((e) => e.type === 'MeterValueRejected');
    if (rejected) return reply.code(422).send({ error: 'MeterValueRejected', reason: rejected.reason, lastMeterWh: rejected.lastMeterWh });
    return reply.code(202).send({ sessionId: req.params.id, meterWh: req.body.meterWh });
  });

  server.post<{ Params: { id: string }; Body: { stopReason?: string; at: string } }>('/sessions/:id/stop', async (req, reply) => {
    const events = await app.uow.run(() => app.charging.stopCharging({ sessionId: req.params.id, stopReason: req.body.stopReason ?? 'Local', at: req.body.at, ...cid(req) }));
    const completed = events.find((e) => e.type === 'ChargingCompleted');
    return reply.code(200).send({ sessionId: req.params.id, status: 'Completed', energyWh: completed?.energyWh });
  });

  server.post<{ Body: { chargerId: string; connectorId?: string; faultCode: string; at: string } }>('/faults', async (req, reply) => {
    const b = req.body;
    const events = await app.uow.run(() => app.charging.reportFault({ chargerId: b.chargerId, connectorId: b.connectorId, faultCode: b.faultCode, at: b.at, ...cid(req) }));
    const faulted = events.find((e) => e.type === 'ChargerFaulted');
    return reply.code(202).send({ chargerId: b.chargerId, faultCode: b.faultCode, stillEnergized: faulted?.stillEnergized ?? false });
  });

  // ---- Queries ---------------------------------------------------------------------------
  server.get<{ Params: { id: string } }>('/sessions/:id', async (req, reply) => {
    const s = await app.sessions.findById(req.params.id);
    if (!s) return reply.code(404).send({ error: 'NotFound', sessionId: req.params.id });
    return { sessionId: s.sessionId, connectorId: s.connectorId, status: s.status, idTag: s.idTag, meterStartWh: s.meterStartWh, lastMeterWh: s.lastMeterWh, startedAt: s.startedAt };
  });

  server.get<{ Querystring: { chargerId?: string } }>('/work-orders', async (req) => {
    const all = app.workOrders.all();
    return all
      .filter((w) => !req.query.chargerId || w.chargerId === req.query.chargerId)
      .map((w) => ({ workOrderId: w.workOrderId, chargerId: w.chargerId, connectorId: w.connectorId, faultCode: w.faultCode, status: w.status, technicianId: w.technicianId, duplicateReportCount: w.duplicateReportCount, outcome: w.outcome }));
  });

  server.get<{ Querystring: { sessionId: string } }>('/invoices', async (req) => {
    const d = app.billing.draftFor(req.query.sessionId);
    return d ? [d] : [];
  });

  // ---- OCPP entry point: the simulator (or a real charger over WebSocket) lands here -------
  server.post<{ Params: { chargerId: string }; Body: OcppCall }>('/ocpp/:chargerId', async (req, reply) => {
    const acl = acls.get(req.params.chargerId) ?? new OcppAcl(req.params.chargerId, app.charging, app.authorization);
    acls.set(req.params.chargerId, acl);
    try {
      return await app.uow.run(() => acl.handle(req.body, cid(req)));
    } catch (e) {
      if (e instanceof InvalidStateError) return reply.code(409).send({ error: 'InvalidState', message: (e as Error).message });
      return reply.code(400).send({ error: 'BadRequest', message: (e as Error).message });
    }
  });

  // ---- AssetOps: the technician reports "repair verified" ---------------------------------
  server.post<{ Params: { id: string }; Body: { outcome: string; at: string; correlationId?: string } }>('/work-orders/:id/close', async (req) => {
    const wo = await app.uow.run(() => app.faults.closeWorkOrder(req.params.id, req.body.outcome, req.body.at, req.body.correlationId ?? req.id));
    return { workOrderId: wo.workOrderId, status: wo.status, outcome: wo.outcome };
  });

  // ---- Parking stub (R6): releasing never depends on HQ ------------------------------------
  server.post<{ Body: { parkingSessionId: string; plate: string; at: string } }>('/parking/enter', async (req, reply) => {
    const p = new ParkingSession(req.body.parkingSessionId, req.body.plate, req.body.at);
    parking.set(p.parkingSessionId, p);
    return reply.code(201).send({ parkingSessionId: p.parkingSessionId, plate: p.plate, status: p.status });
  });
  server.post<{ Params: { id: string }; Body: { releasedBy: string; at: string } }>('/parking/:id/manual-release', async (req, reply) => {
    const p = parking.get(req.params.id);
    if (!p) return reply.code(404).send({ error: 'NotFound' });
    p.manualRelease(req.body.releasedBy, req.body.at);
    return { parkingSessionId: p.parkingSessionId, status: p.status, events: p.pullEvents().map((e) => e.type) };
  });
  server.get<{ Params: { id: string } }>('/parking/:id', async (req, reply) => {
    const p = parking.get(req.params.id);
    return p ? { parkingSessionId: p.parkingSessionId, plate: p.plate, status: p.status } : reply.code(404).send({ error: 'NotFound' });
  });

  // ---- Relay & admin (MVP1 only; no auth) -------------------------------------------------
  server.post('/relay', async () => ({ published: await app.relayAll() }));
  server.get<{ Querystring: { type: string } }>('/admin/outbox', async (req) => app.bus.deliveredOfType(req.query.type));
  server.post<{ Params: { eventId: string } }>('/admin/redeliver/:eventId', async (req) => {
    await app.bus.redeliver(req.params.eventId);
    return { redelivered: req.params.eventId };
  });

  server.setErrorHandler((err: Error, req, reply) => {
    if (err instanceof InvalidStateError) return reply.code(409).send({ error: 'InvalidState', message: err.message });
    if (/Unknown session|Unknown work order/.test(err.message)) return reply.code(404).send({ error: 'NotFound', message: err.message });
    req.log.error({ err, correlationId: req.id }, 'unhandled');
    return reply.code(500).send({ error: 'Internal', message: err.message });
  });
  return server;
}

No business rules here; R1-R4 live in the aggregate."""

from typing import Any

from fastapi import FastAPI, Request, Response
from fastapi.responses import JSONResponse
from pydantic import BaseModel

from adapters.ocpp.ocpp_acl import OcppAcl
from app import App
from charging.application.commands import ReportFault, ReportMeterValue, StartCharging, StopCharging
from charging.domain.errors import InvalidStateError
from charging.domain.events import ChargingCompleted, ChargingStartRejected, ChargerFaulted, MeterValueRejected
from parking.parking_session import ParkingSession


class StartBody(BaseModel):
    connectorId: str
    idTag: str
    at: str
    sessionId: str | None = None
    meterStartWh: int = 0


class MeterBody(BaseModel):
    meterWh: int
    at: str


class StopBody(BaseModel):
    at: str
    stopReason: str = "Local"


class FaultBody(BaseModel):
    chargerId: str
    faultCode: str
    at: str
    connectorId: str | None = None


class CloseBody(BaseModel):
    outcome: str
    at: str
    correlationId: str | None = None


class EnterBody(BaseModel):
    parkingSessionId: str
    plate: str
    at: str


class ReleaseBody(BaseModel):
    releasedBy: str
    at: str


def create_app(app: App) -> FastAPI:
    api = FastAPI(title="EV Charge Ops MVP1")
    acls: dict[str, OcppAcl] = {app.charger_id: app.acl}
    parking: dict[str, ParkingSession] = {}

    def cid(request: Request) -> dict[str, str]:
        c = request.headers.get("x-correlation-id") or str(uuid.uuid4())
        return {"correlation_id": c, "causation_id": c}

    @api.exception_handler(InvalidStateError)
    def invalid_state(_: Request, exc: InvalidStateError) -> Response:
        return JSONResponse(status_code=409, content={"error": "InvalidState", "message": str(exc)})

    @api.exception_handler(KeyError)
    def not_found(_: Request, exc: KeyError) -> Response:
        return JSONResponse(status_code=404, content={"error": "NotFound", "message": str(exc)})

    @api.get("/health")
    def health() -> dict:
        return {"ok": True, "pendingOutbox": len(app.charging_outbox.pending()) + len(app.ops_outbox.pending())}

    # ---- Charging commands -----------------------------------------------------------------
    @api.post("/sessions/start", status_code=201)
    def start(body: StartBody, request: Request, response: Response) -> dict:
        session_id = body.sessionId or f"S-{uuid.uuid4().hex[:8]}"
        cmd = StartCharging(session_id=session_id, connector_id=body.connectorId, id_tag=body.idTag, meter_start_wh=body.meterStartWh, at=body.at, **cid(request))
        events = app.uow.run(lambda: app.charging.start_charging(cmd))
        rejected = next((e for e in events if isinstance(e, ChargingStartRejected)), None)
        if rejected:
            response.status_code = 403 if rejected.reason == "Unauthorized" else 409
            return {"error": "ChargingStartRejected", "reason": rejected.reason}
        return {"sessionId": session_id, "connectorId": body.connectorId, "status": "Charging"}

    @api.post("/sessions/{session_id}/meter", status_code=202)
    def meter(session_id: str, body: MeterBody, request: Request, response: Response) -> dict:
        cmd = ReportMeterValue(session_id=session_id, meter_wh=body.meterWh, at=body.at, **cid(request))
        events = app.uow.run(lambda: app.charging.report_meter_value(cmd))
        rejected = next((e for e in events if isinstance(e, MeterValueRejected)), None)
        if rejected:
            response.status_code = 422
            return {"error": "MeterValueRejected", "reason": rejected.reason, "lastMeterWh": rejected.last_meter_wh}
        return {"sessionId": session_id, "meterWh": body.meterWh}

    @api.post("/sessions/{session_id}/stop")
    def stop(session_id: str, body: StopBody, request: Request) -> dict:
        cmd = StopCharging(session_id=session_id, stop_reason=body.stopReason, at=body.at, **cid(request))
        events = app.uow.run(lambda: app.charging.stop_charging(cmd))
        completed = next((e for e in events if isinstance(e, ChargingCompleted)), None)
        return {"sessionId": session_id, "status": "Completed", "energyWh": completed.energy_wh if completed else None}

    @api.post("/faults", status_code=202)
    def fault(body: FaultBody, request: Request) -> dict:
        cmd = ReportFault(charger_id=body.chargerId, connector_id=body.connectorId, fault_code=body.faultCode, at=body.at, **cid(request))
        events = app.uow.run(lambda: app.charging.report_fault(cmd))
        faulted = next((e for e in events if isinstance(e, ChargerFaulted)), None)
        return {"chargerId": body.chargerId, "faultCode": body.faultCode, "stillEnergized": faulted.still_energized if faulted else False}

    # ---- Queries ---------------------------------------------------------------------------
    @api.get("/sessions/{session_id}")
    def get_session(session_id: str) -> Response:
        s = app.sessions.find_by_id(session_id)
        if s is None:
            return JSONResponse(status_code=404, content={"error": "NotFound", "sessionId": session_id})
        return JSONResponse({"sessionId": s.session_id, "connectorId": s.connector_id, "status": s.status.value, "idTag": s.id_tag,
                             "meterStartWh": s.meter_start_wh, "lastMeterWh": s.last_meter_wh, "startedAt": s.started_at})

    @api.get("/work-orders")
    def work_orders(chargerId: str | None = None) -> list[dict]:
        return [{"workOrderId": w.work_order_id, "chargerId": w.charger_id, "connectorId": w.connector_id, "faultCode": w.fault_code, "status": w.status.value,
                 "technicianId": w.technician_id, "duplicateReportCount": w.duplicate_report_count, "outcome": w.outcome}
                for w in app.work_orders.all() if chargerId is None or w.charger_id == chargerId]

    @api.get("/invoices")
    def invoices(sessionId: str) -> list[dict]:
        d = app.billing.draft_for(sessionId)
        return [{"invoiceId": d.invoice_id, "sessionId": d.session_id, "energyWh": d.energy_wh, "amount": d.amount, "status": d.status}] if d else []

    # ---- OCPP entry point: the simulator (or a real charger over WebSocket) lands here -------
    @api.post("/ocpp/{charger_id}")
    def ocpp(charger_id: str, frame: list[Any], request: Request) -> Response:

        try:

    @api.post("/parking/{parking_id}/manual-release")
    def release(parking_id: str, body: ReleaseBody) -> Response:
        p = parking.get(parking_id)
        if p is None:
            return JSONResponse(status_code=404, content={"error": "NotFound"})
        p.manual_release(body.releasedBy, body.at)
        return JSONResponse({"parkingSessionId": p.parking_session_id, "status": p.status, "events": [e.type for e in p.pull_events()]})

import type { DatabaseSync } from 'node:sqlite';

export interface UnitOfWork {
  run<T>(work: () => Promise<T>): Promise<T>;
}

// One HTTP request = one unit of work = one SQLite transaction (BEGIN ... COMMIT / ROLLBACK).
// node:sqlite is synchronous, but the application service is async, so units are queued one after
// another on this connection (known limitation: single process, no optimistic locking).
export class SqliteUnitOfWork implements UnitOfWork {
  private readonly db: DatabaseSync;
  private queue: Promise<unknown> = Promise.resolve();
  constructor(db: DatabaseSync) {
    this.db = db;
  }

  run<T>(work: () => Promise<T>): Promise<T> {
    const next = this.queue.then(async () => {
      this.db.exec('BEGIN IMMEDIATE');
      try {
        const result = await work();
        this.db.exec('COMMIT');
        return result;
      } catch (e) {
        this.db.exec('ROLLBACK');
        throw e;
      }
    });
    this.queue = next.catch(() => undefined);
    return next;
  }
}

export const noTransaction: UnitOfWork = { run: (work) => work() }; // in-memory adapters

import type { App } from '../app.ts';

// Background relay: drains every outbox every `intervalMs`. Failures are logged, rows stay pending -> retried.
export function startRelayLoop(app: App, intervalMs: number, log: (msg: string, meta?: Record<string, unknown>) => void): () => void {
  let running = false;
  const tick = async () => {
    if (running) return;
    running = true;
    try {
      for (const relay of app.relays) {
        for (const event of await relay['outbox'].pending()) {
          await app.bus.publish(event);
          await relay['outbox'].markPublished(event.eventId);
          log('relay.published', { eventId: event.eventId, type: event.type, correlationId: event.correlationId });
        }
      }
    } catch (e) {
      log('relay.failed', { error: (e as Error).message }); // nothing marked -> at-least-once
    } finally {
      running = false;
    }
  };
  const timer = setInterval(tick, intervalMs);
  return () => clearInterval(timer);
}

        self._lock = threading.Lock()

            try:
                result = work()
                self._conn.execute("COMMIT")
                return result
            except BaseException:
                self._conn.execute("ROLLBACK")
                raise

def start_relay_loop(app: App, interval_s: float) -> threading.Event:

    return stop

E2E 驅動程式：從 in-memory 改成打 HTTP

原版 scripts/e2e/run.mjs 直接 import buildApp。改成 HTTP 版只換三件事：輸入變 fetch('/ocpp/CP-A12', …)（frame 直接用 sim.mjs 的 frames()）、期望狀態變 GET /sessions/S-991、/invoices、/work-orders、/parking/P-441，「relay 跑」變成輪詢 GET /health 直到 pendingOutbox === 0（背景 relay 迴圈在跑）。人的動作（14:33 放行、18:18 關單）各要一個端點。18:12 的重送靠 /admin/redeliver——MVP 的除錯端點，沒認證，列進限制。斷言清單一字不改。

#!/usr/bin/env node
// Day 6 B3: the same afternoon as scripts/e2e/run.mjs, but driven over HTTP against a running MVP1
// (SQLite + relay loop). Every call carries one correlationId so the logs can be grepped end to end.
//   API_URL=http://localhost:3000 node scripts/e2e/run-http.mjs
import { frames } from '../ocpp-sim/sim.mjs';

const BASE = process.env.API_URL ?? 'http://localhost:3000';
const CORRELATION = process.env.CORRELATION_ID ?? 'afternoon-2025-05-20-CP-A12';
const ocpp = Object.fromEntries(frames().map((f) => [f[1], f])); // sim-001 ... sim-010
const rows = [];

async function call(method, path, body) {
  const res = await fetch(BASE + path, { method, headers: { 'content-type': 'application/json', 'x-correlation-id': CORRELATION }, body: body === undefined ? undefined : JSON.stringify(body) });
  const text = await res.text();
  let json; try { json = JSON.parse(text); } catch { json = text; }
  return { status: res.status, json };
}
const get = (path) => call('GET', path).then((r) => r.json);
const post = (path, body) => call('POST', path, body);
const send = (frame) => post('/ocpp/CP-A12', frame);
// "relay ran" = the background relay loop drained every outbox
async function relayed() { for (let i = 0; i < 40; i++) { if ((await get('/health')).pendingOutbox === 0) return; await new Promise((r) => setTimeout(r, 50)); } throw new Error('outbox never drained'); }

async function snapshot(time, input) {
  const s = await get('/sessions/S-991');
  const charging = s.status ? `S-991 ${s.status} (last ${s.lastMeterWh} Wh)` : '—';
  const [draft] = await get('/invoices?sessionId=S-991');
  const billing = draft ? `${draft.invoiceId} ${draft.status} ${draft.energyWh / 1000} kWh / ${draft.amount}` : '—';
  const wos = await get('/work-orders?chargerId=CP-A12');
  const ops = wos.map((w) => `${w.workOrderId} ${w.status}${w.technicianId ? ' → ' + w.technicianId : ''} dup=${w.duplicateReportCount}`).join(', ') || '—';
  const p = await get('/parking/P-441');
  rows.push({ time, input, parking: p.status ? `P-441 ${p.status}` : '—', charging, billing, ops });
}

await post('/parking/enter', { parkingSessionId: 'P-441', plate: 'ABC-1234', at: '2025-05-20T14:02:00+08:00' });
await snapshot('14:02', 'POST /parking/enter ABC-1234');

await snapshot('14:05', `POST /ocpp/CP-A12 StartTransaction TAG-VISITOR-01 → ${blocked.json[2].idTagInfo.status}`);
// 14:13 / 14:22 meter values

Quiz 參考答案

1. OCPP 的 connectorId: 2 與領域的 ConnectorId("CP-A12-2") 誰負責對應？寫進 ChargingSession 會有什麼問題？

參考答案

ACL（connectorIdOf(chargerId, 2)，在 adapters/ocpp/）。寫進聚合有三個問題：領域層知道了 OCPP 的編號方式，違反「OCPP 不得洩入領域層」；換 OCPP 2.0.1（evse.id + connectorId 兩層）時聚合要改，而業務事實「連接器 CP-A12-2」根本沒變；chargerId 不在訊息裡、在連線（URL）上，聚合拿不到它，只能靠 adapter 補。看 OcppAcl.ts 第 47–48 行與 ConnectorIdMapper。

2. 今天你改了幾行 src/charging/domain/ 或 src/charging/application/？不是零的話那些改動代表什麼？

參考答案

理想是零；參考實作的 Charging 是零，AssetOps 的 WorkOrder 多了 12 行 rehydrate()。非零代表 Day 4–5 有漏：聚合沒為持久化準備快照工廠（rehydrate）、或沒暴露足夠的 getter（endedAt、stopReason 就是例子，參考實作選擇讓那兩欄 NULL 而不改聚合）、或處理器把時間寫死用伺服器時鐘。這些不扣分，但要在 README「已知限制」或 retro.md 寫出來——它們是 Day 7 retro 最好的「數字」。

3. 為什麼 outbox 表要和 charging_sessions 在同一個資料庫、同一個交易？分開放會發生什麼？

參考答案

同一交易才有原子性：聚合寫入成功 ⇔ 事件寫入成功，兩者要嘛都在、要嘛都不在。分開放（事件寫到另一個 DB、或直接 bus.publish()）就回到 Day 5 Quiz 1 的兩種不一致：存成功發失敗（Billing 永遠不知道 S-991 結束了，少收）、發成功存失敗（Billing 收到一張 12.4 kWh 的草稿，但 S-991 在資料庫裡還是 Charging，帳對不上）。測試 SQLite unit of work rolls back the outbox when the session save fails 就是在釘這件事；三個 adapter 共用同一個 db 連線是它成立的前提。

4. E2E 的 18:15（本 repo 模擬器是 18:10:08）那步：第二次 StatusNotification(Faulted, GroundFailure) 走過哪幾層、各做了什麼判斷，最後為什麼只有一張工單？

參考答案

HTTP POST /ocpp/CP-A12（解 frame、開交易）→ ACL：status === 'Faulted' 才翻，connectorId 2 → CP-A12-2，errorCode → faultCode，翻成 ReportFault(CP-A12, CP-A12-2, GroundFailure, 18:10:08) → ChargingService.reportFault：沒有進行中會話，仍在一個 idle 聚合上記 ChargerFaulted(stillEnergized=false)，不 persist 會話、只寫 outbox → commit → relay 把新的 charging.charger.faulted.v1（新 eventId）發到 bus → FaultProcessManager.onChargerFaulted：processedEventIds 沒看過這個 id，放行；workOrders.findOpenByFault('CP-A12','GroundFailure') 找到 WO-2208（SQL 有 status <> 'Closed'）→ existing.appendDuplicateReport(18:10:08)（R5）→ save，DuplicateFaultReported 不跨 context，所以 AssetOps outbox 沒有新的 opened 信封 → 只有一張，duplicateReportCount 1。18:12 的重送走到 processedEventIds 就停了，dup 仍 1——那是技術冪等，跟 R5 的業務去重是兩件事。

5. relay 掛掉 10 分鐘再起來，Billing 會多收還是少收 charging.session.completed.v1？哪一層保證結果正確？

參考答案

不會少收：事件在 outbox 裡 published_at IS NULL 等著，relay 回來就送（斷線劇本那 12 行就是證據：4 列 pending → published: 5）。可能多收：at-least-once，例如 relay publish 成功、markPublished 之前掛掉，下次會再送同一個 eventId。保證結果正確的是消費者的冪等：Billing 以 sessionId 去重（drafts.has(sessionId) → 仍一張 INV-778；completed 重複套用是同輸入同結果），AssetOps 以 eventId（processedEventIds）+ chargerId+faultCode（findOpenByFault）去重。broker 與 relay 只負責「至少一次」，正確性從來不在它們身上。

講師手記

教材夠不夠：Day 6 的教材是七天裡最厚的一份（docs/days/day6.md 三個 Block 每一步都有時間、檔名、測試名），而且 repo 給的東西剛好卡在對的位置：ACL 有參考解、模擬器與 E2E 驅動程式現成、docker/init.sql 把表設計都給了。學員真正要自己寫的是 HTTP 層、四個 SQLite adapter、relay 迴圈、Dockerfile、E2E 的 HTTP 版——這五樣 solutions/ 刻意不提供，所以這一頁的參考實作（chip「參考實作 · solutions 不提供」）是講師唯一的對照答案。它們全部真跑過：Node 48 / Python 48 測試綠、兩種語言的 E2E 11 列 8 PASS、斷線劇本通。唯一沒實測的是 Docker build（環境無 daemon），已在頁面上明說。

時間一定會爆，這是七天裡最可能超時的一天。原因具體：（1）B1 的 HTTP 層對沒碰過 Fastify / FastAPI 的學員是 60 分鐘不是 35 分鐘，光是「測試互相污染」（每個測試要 buildServer() 一次）就會吃掉 15 分鐘；（2）B2 的環境坑是真的——Node 的 node:sqlite 需要 22.13+ 且 vitest 讀不到、Python 的 isolation_level、FastAPI 的裸 list——寫這一頁時每一個都踩到了，學員也會；（3）B3 的 E2E 改寫看起來簡單，但 18:12 重送與 18:18 關單需要兩個學員原本沒想到的端點，加上 correlationId 串不起來要回頭補 ACL。實際估計：認真做完三個 Block 需要 5–6 小時，不是 4.5。

砍什麼（按順序）：第一刀砍 Docker（教材已標選配，砍了不影響 DoD）。第二刀砍 WorkOrder 的 SQLite repository（留 in-memory，寫進限制；ChargingSession + Outbox + UoW 就足以證明「核心不動」）。第三刀砍 E2E 的 18:12 重送步驟（redeliver 端點是 MVP 除錯用的，Day 5 的 FaultProcessManager.test 已經釘了同 eventId 重送）。第四刀砍 Postgres 一切。不能砍的：14:05 的 R1 Blocked、18:10:08 的 R5 dup=1、斷線劇本那 12 行、README 的「怎麼跑」與「已知限制」——這四樣是明天 PR reviewer 判斷「這個 MVP 是真的還是假的」的依據。

Docker-less 路徑：完全可行，這一頁本身就是在沒有 Docker daemon 的環境跑出來的。npm start / python src/main.py + DB_PATH 就是「部署」；rubric 明列 Docker 為選配不扣分。講師要做的只有一件事：kickoff 時說清楚「沒有 Docker 的人今天不會少學任何東西」，否則會有學員花 40 分鐘裝 Docker Desktop。
